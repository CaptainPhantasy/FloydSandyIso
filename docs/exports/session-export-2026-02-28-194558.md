func NewTelemetryConsumer(cfg ConsumerConfig) (*TelemetryConsumer, error) {
	ctx, cancel := context.WithCancel(context.Background())
	
	tc := &TelemetryConsumer{
		config:      cfg,
		ctx:         ctx,
		cancel:      cancel,
		logger:      log.New(os.Stdout, "[dnet-xdp] ", log.LstdFlags|log.Lmicroseconds),
		metricsTick: time.NewTicker(cfg.MetricsInterval),
	}

	// Load pre-compiled eBPF object
	spec, err := ebpf.LoadCollectionSpec(cfg.EBPFObjectPath)
	if err != nil {
		cancel()
		return nil, fmt.Errorf("failed to load eBPF spec: %w", err)
	}

	// Load the collection (maps + programs)
	tc.collection, err = ebpf.NewCollection(spec)
	if err != nil {
		cancel()
		return nil, fmt.Errorf("failed to create eBPF collection: %w", err)
	}

	// Get the telemetry_events perf event map
	telemetryMap, ok := tc.collection.Maps["telemetry_events"]
	if !ok {
		tc.collection.Close()
		cancel()
		return nil, fmt.Errorf("telemetry_events map not found in eBPF object")
	}

	// Create perf reader
	tc.perfReader, err = perf.NewReader(telemetryMap, cfg.PerfBufferPages*os.Getpagesize())
	if err != nil {
		tc.collection.Close()
		cancel()
		return nil, fmt.Errorf("failed to create perf reader: %w", err)

	if cfg.OutputFormat == "syslog" && cfg.SyslogAddr != "" {

func (tc *TelemetryConsumer) Start() error {
	tc.logger.Println("Starting telemetry consumer...")
	tc.logger.Printf("Perf buffer: %d pages (%d bytes)", 
		tc.config.PerfBufferPages, 
		tc.config.PerfBufferPages*os.Getpagesize())

	for {
		select {
		case <-tc.ctx.Done():
			return tc.ctx.Err()

		case <-tc.metricsTick.C:
			tc.emitMetrics()

		case record, err := tc.perfReader.Read(); err != nil {
			if errors.Is(err, perf.ErrLostEvents) {
				tc.logger.Printf("WARNING: Lost %d events due to buffer overflow", record.LostSamples)
				continue
			}
			tc.logger.Printf("Error reading from perf buffer: %v", err)

func (tc *TelemetryConsumer) handleEvent(data []byte) {
	if len(data) < 40 { // Minimum size of TelemetryEvent
		tc.metrics.ParseErrors++
		return
	}

	var evt TelemetryEvent
	
	// Manual parsing to ensure correct byte order and alignment
	// Structure: timestamp(8) + event_type(4) + cpu_id(4) + pkt_len(4) + ifindex(4) + 
	//            src_mac(6) + dst_mac(6) + eth_proto(2) + reserved(4) = 42 bytes
	evt.Timestamp = binary.LittleEndian.Uint64(data[0:8])
	evt.EventType = binary.LittleEndian.Uint32(data[8:12])
	evt.CPUID = binary.LittleEndian.Uint32(data[12:16])
	evt.PktLen = binary.LittleEndian.Uint32(data[16:20])
	evt.IfIndex = binary.LittleEndian.Uint32(data[20:24])
	copy(evt.SrcMAC[:], data[24:30])
	copy(evt.DstMAC[:], data[30:36])
	evt.EthProto = binary.LittleEndian.Uint16(data[36:38])

	tc.metrics.TotalEvents++
	tc.metrics.LastTimestamp = evt.Timestamp

	switch evt.EventType {
	case EvtDropMalformed:
		tc.metrics.Drops++
		tc.outputEvent("DROP", &evt)
	case EvtPassValid:
		tc.metrics.Passes++
		if !tc.config.DropOnly {
			tc.outputEvent("PASS", &evt)
		}
	case EvtErrorParse:
		tc.metrics.ParseErrors++
		tc.outputEvent("ERROR", &evt)
	}
}

func (tc *TelemetryConsumer) outputEvent(action string, evt *TelemetryEvent) {

			evt.CPUID,
			evt.IfIndex,

	}
